## 자동화 스크립트 메모 (재사용)

다음 세션에서 새 템플릿 분석 시 사용할 수 있는 핵심 JS 스니펫.

### 카드 라벨 + 타입 추출 (한 페이지)
```javascript
(() => {
  const cards = document.querySelectorAll('[class*="group/block"]');
  const result = [];
  cards.forEach((card, idx) => {
    const svgs = card.querySelectorAll('svg');
    let typeKey = null;
    // Badge/Card Icon/22 has rect[rx="4"] background
    for (const svg of svgs) {
      const rect = svg.querySelector('rect[rx="4"]');
      if (rect) {
        const paths = svg.querySelectorAll('path');
        if (paths.length > 0) {
          typeKey = (paths[0].getAttribute('d') || '').slice(0, 40);
          break;
        }
      }
    }
    const label = card.querySelector('input.truncate')?.value || 
                  card.querySelector('.truncate')?.textContent.trim() || '';
    const isPinned = svgs.length < 3;
    result.push({ idx, label, typeKey, isPinned });
  });
  return JSON.stringify(result);
})()
```

### 멀티페이지 패널 열기
```javascript
(() => {
  const toggle = Array.from(document.querySelectorAll('div.bg-grey-900')).find(el => {
    const r = el.getBoundingClientRect();
    return r.left < 5 && r.width < 30 && r.height > 0;
  });
  if (toggle) toggle.click();
})()
```

### 페이지 N 썸네일 클릭 (window.__pageToClick에 페이지 번호 사전 설정)
```javascript
(() => {
  const pageNum = window.__pageToClick;
  const labels = Array.from(document.querySelectorAll('*')).filter(el => {
    const t = el.textContent.trim();
    return t === String(pageNum) && el.children.length === 0;
  });
  const label = labels.find(el => {
    const rect = el.getBoundingClientRect();
    return rect.top > 380 && rect.top < 420 && rect.left < 700;
  });
  if (!label) return;
  let target = label.previousElementSibling;
  while (target && target.tagName !== 'IMG') {
    const img = target.querySelector('img');
    if (img) { target = img; break; }
    target = target.parentElement;
    if (!target) break;
  }
  if (!target) return;
  const rect = target.getBoundingClientRect();
  ['mousedown', 'mouseup', 'click'].forEach(type => {
    target.dispatchEvent(new MouseEvent(type, {
      bubbles: true, cancelable: true, view: window,
      clientX: rect.left + rect.width/2, clientY: rect.top + rect.height/2, button: 0
    }));
  });
})()
```
